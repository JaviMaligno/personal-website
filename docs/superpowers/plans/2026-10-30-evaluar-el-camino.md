```python
"""Expression trees that never simplify on their own.

sympy auto-cancels x*y/x and (x+1)/(x+1) at construction time, which would let an
agent cancel a factor without any rule, and without any side condition. So the
state of the rewrite world is a plain tuple tree; sympy is only consulted to check
conditions and to compute what expand/factor produce.

Node shapes:
    ("num", int) ("sym", str) ("add", (t, ...)) ("mul", (t, ...))
    ("div", num, den) ("pow", base, int) ("sqrt", t) ("log", t) ("neg", t)
"""
from __future__ import annotations

import ast

import sympy

PREC = {"add": 1, "mul": 2, "div": 2, "neg": 3, "pow": 4,
        "num": 5, "sym": 5, "sqrt": 5, "log": 5}
ASSUME = {"positive": {"positive": True},
          "nonzero": {"real": True, "nonzero": True},
          "real": {"real": True}}


def to_str(t) -> str:
    k = t[0]
    if k == "num":
        return str(t[1])
    if k == "sym":
        return t[1]
    if k in ("sqrt", "log"):
        return f"{k}({to_str(t[1])})"
    if k == "neg":
        return "-" + _wrap(t[1], 3)
    if k == "pow":
        return f"{_wrap(t[1], 5)}**{t[2]}"
    if k == "mul":
        return "*".join(_wrap(c, 2.5) for c in t[1])
    if k == "div":
        return f"{_wrap(t[1], 2)}/{_wrap(t[2], 5)}"
    out = to_str(t[1][0])
    for c in t[1][1:]:
        out += f" - {_wrap(c[1], 1.5)}" if c[0] == "neg" else f" + {_wrap(c, 1.5)}"
    return out


def _wrap(t, need) -> str:
    s = to_str(t)
    return f"({s})" if PREC[t[0]] < need else s


def parse(s: str):
    return _from_ast(ast.parse(s.strip(), mode="eval").body)


def _from_ast(n):
    if isinstance(n, ast.Constant) and isinstance(n.value, int):
        return ("num", n.value)
    if isinstance(n, ast.Name):
        return ("sym", n.id)
    if isinstance(n, ast.UnaryOp) and isinstance(n.op, ast.USub):
        return ("neg", _from_ast(n.operand))
    if isinstance(n, ast.BinOp):
        a, b = _from_ast(n.left), _from_ast(n.right)
        if isinstance(n.op, ast.Add):
            return _flat("add", [a, b])
        if isinstance(n.op, ast.Sub):
            return _flat("add", [a, ("neg", b)])
        if isinstance(n.op, ast.Mult):
            return _flat("mul", [a, b])
        if isinstance(n.op, ast.Div):
            return ("div", a, b)
        if isinstance(n.op, ast.Pow) and b[0] == "num":
            return ("pow", a, b[1])
    if (isinstance(n, ast.Call) and isinstance(n.func, ast.Name)
            and n.func.id in ("sqrt", "log") and len(n.args) == 1):
        return (n.func.id, _from_ast(n.args[0]))
    raise ValueError(f"unsupported expression: {ast.unparse(n)}")


def _flat(kind, items):
    out = []
    for it in items:
        out.extend(it[1] if it[0] == kind else [it])
    return (kind, tuple(out))


def _key(kind, c):
    # Sums read "x + 1" and "x - 1"; products put the number first ("2*x").
    if kind == "add":
        return (c[0] == "num", c[0] == "neg", to_str(c))
    return (c[0] not in ("num", "neg"), to_str(c))


def norm(t):
    k = t[0]
    if k in ("add", "mul"):
        kids = _flat(k, [norm(c) for c in t[1]])[1]
        return (k, tuple(sorted(kids, key=lambda c: _key(k, c))))
    if k == "div":
        return ("div", norm(t[1]), norm(t[2]))
    if k == "pow":
        return ("pow", norm(t[1]), t[2])
    if k in ("sqrt", "log", "neg"):
        return (k, norm(t[1]))
    return t


def subterms(t):
    yield t
    k = t[0]
    if k in ("add", "mul"):
        for c in t[1]:
            yield from subterms(c)
    elif k == "div":
        yield from subterms(t[1])
        yield from subterms(t[2])
    elif k in ("pow", "sqrt", "log", "neg"):
        yield from subterms(t[1])


def replace(t, old, new):
    """Replace the first occurrence (preorder) of `old`. Returns (tree, replaced)."""
    if t == old:
        return new, True
    k = t[0]
    if k in ("add", "mul"):
        kids = list(t[1])
        for i, c in enumerate(kids):
            r, ok = replace(c, old, new)
            if ok:
                kids[i] = r
                return (k, tuple(kids)), True
        return t, False
    if k == "div":
        r, ok = replace(t[1], old, new)
        if ok:
            return ("div", r, t[2]), True
        r, ok = replace(t[2], old, new)
        return (("div", t[1], r), True) if ok else (t, False)
    if k == "pow":
        r, ok = replace(t[1], old, new)
        return ("pow", r, t[2]), ok
    if k in ("sqrt", "log", "neg"):
        r, ok = replace(t[1], old, new)
        return (k, r), ok
    return t, False


def to_sympy(t, assumptions: dict | None = None):
    loc = {name: sympy.Symbol(name, **ASSUME[a]) for name, a in (assumptions or {}).items()}
    loc.update(sqrt=sympy.sqrt, log=sympy.log)
    return sympy.sympify(to_str(t), locals=loc)


def from_sympy(e):
    return norm(parse(str(e)))


def canon(v) -> str:
    """Canonical spelling of an argument; non-expressions are kept as given."""
    try:
        return to_str(norm(parse(str(v))))
    except (ValueError, SyntaxError):
        return str(v)
```

```python
"""The exploratory world: turn a start expression into a target by applying rules.

Like a sloppy mathematician, the world applies a rule even when its side condition
does not hold: it records the step as invalid instead of refusing it. Refusing would
leave nothing for the evaluators to catch.
"""
from __future__ import annotations

import sympy

from expr import canon, from_sympy, norm, parse, replace, subterms, to_str, to_sympy

CHECK = {"nonzero": "is_nonzero", "nonnegative": "is_nonnegative", "positive": "is_positive"}

_SUB = {"type": "object", "properties": {"sub": {"type": "string"}}, "required": ["sub"]}
TOOLS = [
    {"name": "expand", "description": "Expand a subexpression that contains no '/'. No side condition.",
     "schema": _SUB},
    {"name": "factor", "description": "Factor a subexpression that contains no '/'. No side condition.",
     "schema": _SUB},
    {"name": "cancel", "description": "In the fraction `sub`, cancel one occurrence of `factor` from "
                                      "numerator and denominator. Valid only if factor != 0.",
     "schema": {"type": "object", "properties": {"sub": {"type": "string"}, "factor": {"type": "string"}},
                "required": ["sub", "factor"]}},
    {"name": "sqrt_square", "description": "Rewrite sqrt(b**2) as b. Valid only if b >= 0.", "schema": _SUB},
    {"name": "log_product", "description": "Rewrite log(a*b*...) as log(a) + log(b) + ... "
                                           "Valid only if every factor > 0.", "schema": _SUB},
    {"name": "prove_nonzero", "description": "Check under the hypotheses that `sub` != 0.", "schema": _SUB},
    {"name": "prove_nonnegative", "description": "Check under the hypotheses that `sub` >= 0.", "schema": _SUB},
    {"name": "prove_positive", "description": "Check under the hypotheses that `sub` > 0.", "schema": _SUB},
    {"name": "submit", "description": "Declare the derivation finished.",
     "schema": {"type": "object", "properties": {}}},
]


def holds(t, cond, assumptions) -> bool:
    """True only if sympy can prove the condition under the task's hypotheses."""
    return getattr(to_sympy(t, assumptions), CHECK[cond]) is True


def token(tool, args) -> str:
    return f"{tool}(" + ", ".join(f"{k}={canon(args[k])}" for k in sorted(args)) + ")"


def _drop(t, f):
    """Remove one factor `f` from product `t`; None if `f` is not a factor."""
    if t == f:
        return ("num", 1)
    if t[0] == "pow" and t[1] == f:
        return t[1] if t[2] == 2 else ("pow", t[1], t[2] - 1)
    if t[0] == "mul":
        for i, c in enumerate(t[1]):
            r = _drop(c, f)
            if r is not None:
                kids = [k for j, k in enumerate(t[1]) if j != i] + ([] if r == ("num", 1) else [r])
                return kids[0] if len(kids) == 1 else ("mul", tuple(kids))
    return None


class RewriteWorld:
    def __init__(self, start: str, target: str, assumptions: dict, kind: str = "solvable"):
        self.state = norm(parse(start))
        self.goal = norm(parse(target))
        self.assumptions = assumptions
        self.kind = kind
        self.steps: list[dict] = []
        self.done = False

    def call(self, tool: str, args: dict) -> str:
        try:
            result, changed, valid = self._apply(tool, args)
        except (ValueError, SyntaxError, KeyError, TypeError) as exc:
            result, changed, valid = f"error: {exc}", False, None
        progress = changed or (tool.startswith("prove_") and result == "proved")
        self.steps.append({"tool": tool, "args": args, "result": result, "changed": changed,
                           "valid": valid, "progress": progress, "token": token(tool, args),
                           "state": to_str(self.state)})
        return result

    def final_ok(self) -> bool:
        return self.state == self.goal

    def truth(self) -> bool:
        no_invalid = all(s["valid"] is not False for s in self.steps)
        return no_invalid if self.kind == "trap" else no_invalid and self.final_ok()

    def _find(self, s):
        sub = norm(parse(s))
        if sub not in set(subterms(self.state)):
            raise ValueError(f"'{s}' is not a subexpression of the current expression")
        return sub

    def _set(self, sub, new):
        self.state = norm(replace(self.state, sub, norm(new))[0])

    def _apply(self, tool, args):
        if tool == "submit":
            self.done = True
            return f"submitted: {to_str(self.state)}", False, None
        if tool not in {t["name"] for t in TOOLS}:
            raise ValueError(f"unknown tool {tool}")
        sub = self._find(args["sub"])
        if tool.startswith("prove_"):
            ok = holds(sub, tool.removeprefix("prove_"), self.assumptions)
            return ("proved" if ok else "not proved"), False, None
        if tool in ("expand", "factor"):
            if any(t[0] == "div" for t in subterms(sub)):
                raise ValueError(f"{tool} only applies to subexpressions without '/'")
            new = from_sympy((sympy.expand if tool == "expand" else sympy.factor)(to_sympy(sub)))
            if new == sub:
                raise ValueError(f"{tool} does not change '{args['sub']}'")
            self._set(sub, new)
            return to_str(self.state), True, None
        if tool == "cancel":
            if sub[0] != "div":
                raise ValueError("cancel needs a fraction")
            f = norm(parse(args["factor"]))
            num, den = _drop(sub[1], f), _drop(sub[2], f)
            if num is None or den is None:
                raise ValueError(f"'{args['factor']}' is not a factor of both numerator and denominator")
            self._set(sub, num if den == ("num", 1) else ("div", num, den))
            return to_str(self.state), True, holds(f, "nonzero", self.assumptions)
        if tool == "sqrt_square":
            if not (sub[0] == "sqrt" and sub[1][0] == "pow" and sub[1][2] == 2):
                raise ValueError("sqrt_square needs sqrt(b**2)")
            b = sub[1][1]
            self._set(sub, b)
            return to_str(self.state), True, holds(b, "nonnegative", self.assumptions)
        if not (sub[0] == "log" and sub[1][0] == "mul"):  # log_product
            raise ValueError("log_product needs log(a*b*...)")
        factors = sub[1][1]
        self._set(sub, ("add", tuple(("log", f) for f in factors)))
        return to_str(self.state), True, all(holds(f, "positive", self.assumptions) for f in factors)
```

```python
"""The procedural world: a customer onboarding flow whose correct path is known a priori."""
from __future__ import annotations

import json

THRESHOLD = 10000
POLICY = f"""Customer onboarding policy
1. Before any decision (approve or reject), verify the customer's identity and check sanctions.
2. If the requested amount is above {THRESHOLD} EUR, assess risk before deciding. Risk does not matter at or below {THRESHOLD} EUR.
3. Approve only if identity is verified, there is no sanctions hit and, when required, risk is low. Otherwise reject.
4. Make exactly one decision, then record it with log_decision."""
PROGRESS = ("verify_identity", "check_sanctions", "assess_risk", "approve", "reject", "log_decision")
DECISIONS = ("approve", "reject")

_NONE = {"type": "object", "properties": {}}
TOOLS = [
    {"name": "get_customer", "description": "Read the application: customer name and requested amount.",
     "schema": _NONE},
    {"name": "get_policy", "description": "Read the onboarding policy.", "schema": _NONE},
    {"name": "verify_identity", "description": "Run the identity check. Returns verified or failed.",
     "schema": _NONE},
    {"name": "check_sanctions", "description": "Screen against sanctions lists. Returns clear or hit.",
     "schema": _NONE},
    {"name": "assess_risk", "description": "Run the risk assessment. Returns low or high.", "schema": _NONE},
    {"name": "approve", "description": "Approve the application.", "schema": _NONE},
    {"name": "reject", "description": "Reject the application.",
     "schema": {"type": "object", "properties": {"reason": {"type": "string"}}}},
    {"name": "log_decision", "description": "Record the decision. Ends the case.", "schema": _NONE},
]


def token(tool, args) -> str:
    return tool


def correct_decision(c) -> str:
    risky = c["amount"] > THRESHOLD and c["risk"] == "high"
    return "approve" if c["identity_ok"] and not c["sanctions_hit"] and not risky else "reject"


def reference_paths(c) -> list[list[str]]:
    risk = ["assess_risk"] if c["amount"] > THRESHOLD else []
    tail = risk + [correct_decision(c), "log_decision"]
    return [["verify_identity", "check_sanctions"] + tail,
            ["check_sanctions", "verify_identity"] + tail]


class OnboardingWorld:
    def __init__(self, customer: dict):
        self.c = customer
        self.steps: list[dict] = []
        self.done = False

    def call(self, tool: str, args: dict) -> str:
        c = self.c
        results = {
            "get_customer": lambda: json.dumps({"id": c["id"], "name": c["name"], "amount": c["amount"]}),
            "get_policy": lambda: POLICY,
            "verify_identity": lambda: "verified" if c["identity_ok"] else "failed",
            "check_sanctions": lambda: "hit" if c["sanctions_hit"] else "clear",
            "assess_risk": lambda: c["risk"],
            "approve": lambda: "approved",
            "reject": lambda: "rejected",
            "log_decision": lambda: "logged",
        }
        seen = {s["tool"] for s in self.steps if s["progress"]}
        result = results[tool]() if tool in results else f"error: unknown tool {tool}"
        if tool == "log_decision":
            self.done = True
        self.steps.append({"tool": tool, "args": args, "result": result,
                           "progress": tool in PROGRESS and tool not in seen,
                           "token": token(tool, args)})
        return result

    def _decisions(self):
        return [(i, s["tool"]) for i, s in enumerate(self.steps) if s["tool"] in DECISIONS]

    def final_ok(self) -> bool:
        d = self._decisions()
        logged = any(s["tool"] == "log_decision" for s in self.steps)
        return bool(d) and d[-1][1] == correct_decision(self.c) and logged

    def truth(self) -> bool:
        d = self._decisions()
        if len(d) != 1 or d[0][1] != correct_decision(self.c):
            return False
        i = d[0][0]
        before = {s["tool"] for s in self.steps[:i]}
        required = {"verify_identity", "check_sanctions"}
        if self.c["amount"] > THRESHOLD:
            required.add("assess_risk")
        logged_after = any(s["tool"] == "log_decision" for s in self.steps[i + 1:])
        return required <= before and logged_after
```

```python
"""Reference derivations for the rewrite world, found by search rather than written by hand.

Same criterion as CORE's golden paths: shortest, loop-free, no invalid step. Each
conditional step is preceded by the proof of its condition, the way a careful
derivation would be written.
"""
from __future__ import annotations

from functools import lru_cache

from expr import norm, parse, subterms, to_str
from rewrite import RewriteWorld

RULES = ("expand", "factor", "sqrt_square", "log_product")


def candidate_moves(state):
    for sub in dict.fromkeys(subterms(state)):
        s = to_str(sub)
        for tool in RULES:
            yield tool, {"sub": s}
        if sub[0] == "div":
            for f in dict.fromkeys(subterms(sub[1])):
                yield "cancel", {"sub": s, "factor": to_str(f)}


@lru_cache(maxsize=None)
def _successors(state, target, assumptions_items, allow_invalid):
    out = []
    for tool, args in candidate_moves(state):
        w = RewriteWorld(to_str(state), target, dict(assumptions_items))
        w.call(tool, args)
        step = w.steps[-1]
        if step["changed"] and (allow_invalid or step["valid"] is not False):
            out.append(((tool, args), w.state))
    return out


def successors(state, task, allow_invalid=False):
    return _successors(state, task["target"], tuple(sorted(task["assumptions"].items())), allow_invalid)


def shortest(task, max_depth=6, limit=2, allow_invalid=False):
    start, goal = norm(parse(task["start"])), norm(parse(task["target"]))
    frontier, seen, found = [(start, [])], {start}, []
    for _ in range(max_depth):
        nxt = []
        for state, path in frontier:
            for move, new in successors(state, task, allow_invalid):
                if new == goal:
                    found.append(path + [move])
                elif new not in seen:
                    seen.add(new)
                    nxt.append((new, path + [move]))
        if found:
            return found[:limit]
        frontier = nxt
    return []


def count_derivations(task, max_len):
    """Number of valid derivations (walks to the goal) of length <= max_len."""
    goal = norm(parse(task["target"]))

    @lru_cache(maxsize=None)
    def paths(state, remaining):
        total = 0
        for _, new in successors(state, task):
            if new == goal:
                total += 1
            elif remaining > 1:
                total += paths(new, remaining - 1)
        return total

    return paths(norm(parse(task["start"])), max_len)


def with_proofs(path):
    out = []
    for tool, args in path:
        if tool == "cancel":
            out.append(("prove_nonzero", {"sub": args["factor"]}))
        elif tool == "sqrt_square":
            out.append(("prove_nonnegative", {"sub": to_str(norm(parse(args["sub"]))[1][1])}))
        elif tool == "log_product":
            for f in norm(parse(args["sub"]))[1][1]:
                out.append(("prove_positive", {"sub": to_str(f)}))
        out.append((tool, args))
    return out
```

```python
"""The four evaluators, from least to most specification of the path.

E0 final state; E1 + forbidden actions; E2 + ordering obligations; E3 + similarity
to a reference path (CORE's Path Correctness). E1 and E2 read hand-written per-task
rules; they are deliberately coarse and are not the semantic truth.
"""
from __future__ import annotations

from expr import canon


def levenshtein(a, b) -> int:
    prev = list(range(len(b) + 1))
    for i, x in enumerate(a, 1):
        cur = [i]
        for j, y in enumerate(b, 1):
            cur.append(min(prev[j] + 1, cur[j - 1] + 1, prev[j - 1] + (x != y)))
        prev = cur
    return prev[-1]


def nld(a, b) -> float:
    ld = levenshtein(a, b)
    return 0.0 if ld == 0 else 2 * ld / (len(a) + len(b) + ld)


def pc(tokens, refs) -> float:
    return max(1 - nld(tokens, r) for r in refs)


def condensed(steps) -> list[str]:
    return [s["token"] for s in steps if s["progress"]]


def _executed(s) -> bool:
    return not str(s["result"]).startswith("error")


def _forbidden(s, rule) -> bool:
    return (s["tool"] == rule["tool"] and _executed(s)
            and all(canon(s["args"].get(k, "")) == canon(v) for k, v in rule.get("args", {}).items()))


def _out_of_order(steps, rule) -> bool:
    for i, s in enumerate(steps):
        if s["tool"] == rule["then"] and s["progress"]:
            if not any(p["tool"] == rule["first"] and p["progress"] for p in steps[:i]):
                return True
    return False


def e0(trace) -> bool:
    return bool(trace["final_ok"])


def e1(trace, rules) -> bool:
    forbids = [r for r in rules if r["kind"] == "forbid"]
    return e0(trace) and not any(_forbidden(s, r) for s in trace["steps"] for r in forbids)


def e2(trace, rules) -> bool:
    befores = [r for r in rules if r["kind"] == "before"]
    return e1(trace, rules) and not any(_out_of_order(trace["steps"], r) for r in befores)


def e3(trace, refs, tau=0.8):
    if not refs:
        return None
    return e0(trace) and pc(condensed(trace["steps"]), refs) >= tau
```

"""Validate the hand-written tasks and attach their reference paths.

    uv run python build_tasks.py

Fails (exit 1) if a solvable task has fewer than two derivations of length
<= shortest + 2, or if a trap task turns out to have a valid derivation.
"""
from __future__ import annotations

import json
import sys
from pathlib import Path

import onboarding
import rewrite
from references import count_derivations, shortest, with_proofs

HERE = Path(__file__).parent


def build_rewrite(src):
    out, bad = [], []
    for t in src:
        if t["kind"] == "solvable":
            paths = shortest(t)
            n = count_derivations(t, len(paths[0]) + 2) if paths else 0
            ok = bool(paths) and n >= 2
            refs = [with_proofs(p) for p in paths]
            print(f"{t['id']} solvable shortest={len(paths[0]) if paths else None} derivations<=+2={n} {'OK' if ok else 'FAIL'}")
        else:
            ok = not shortest(t) and bool(shortest(t, allow_invalid=True))
            refs = []
            print(f"{t['id']} trap {'OK' if ok else 'FAIL'}")
        if not ok:
            bad.append(t["id"])
        out.append({**t, "references": [[{"tool": a, "args": b} for a, b in p] for p in refs],

```python
"""A minimal tool-use loop: ask the model, run its tool calls in the world, repeat."""
from __future__ import annotations

import transport


def run_episode(model, world, system, user, tools, max_calls=30, call=None):
    call = call or transport.call
    msgs = [{"role": "user", "text": user}]
    usage, calls = [], 0
    while True:
        if world.done:
            return {"stop": "done", "calls": calls, "usage": usage}
        if calls >= max_calls:
            return {"stop": "budget", "calls": calls, "usage": usage}
        r = call(model, system, msgs, tools)
        if r["status"] != "ok":
            return {"stop": "error", "error": r["error"], "calls": calls, "usage": usage}
        usage.append(r.get("usage", {}))
        if not r["tool_calls"]:
            return {"stop": "no_tool_call", "calls": calls, "usage": usage, "final_text": r["text"]}
        msgs.append({"role": "assistant", "text": r["text"], "tool_calls": r["tool_calls"],
                     "raw_kind": r["raw_kind"], "raw": r["raw"]})
        for c in r["tool_calls"]:
            # Every tool call needs a result, even the ones past the end of the episode.
            if world.done or calls >= max_calls:
                content = "not executed: the episode is over"
            else:
                content = world.call(c["name"], c["input"] or {})
                calls += 1
            msgs.append({"role": "tool", "id": c["id"], "name": c["name"], "content": content})
```

```python
"""Expose one world as an MCP server over stdio, for agents driven by the Claude Code CLI.

    python mcp_world.py <episode.json> <steps.jsonl>

episode.json holds {"world", "task", "max_calls"}. Every executed tool call is appended
to steps.jsonl as {"tool", "args"}; the runner replays them into a fresh world
afterwards, so this process never has to report scores.
"""
from __future__ import annotations

import json
import sys
from pathlib import Path

sys.path.insert(0, str(Path(__file__).parent))

import onboarding  # noqa: E402
import rewrite  # noqa: E402


def make_world(world, task):
    if world == "rewrite":
        w = rewrite.RewriteWorld(task["start"], task["target"], task["assumptions"], kind=task["kind"])
        return w, rewrite.TOOLS
    return onboarding.OnboardingWorld(task["customer"]), onboarding.TOOLS


def handle(msg, world, tools, log, state):
    method, mid = msg.get("method"), msg.get("id")
    if mid is None:  # notification
        return None
    if method == "initialize":
        version = (msg.get("params") or {}).get("protocolVersion", "2025-06-18")
        result = {"protocolVersion": version, "capabilities": {"tools": {}},
                  "serverInfo": {"name": "world", "version": "0.1.0"}}
    elif method == "tools/list":
        result = {"tools": [{"name": t["name"], "description": t["description"], "inputSchema": t["schema"]}
                            for t in tools]}
    elif method == "tools/call":
        name = msg["params"]["name"]
        args = msg["params"].get("arguments") or {}
        if world.done or state["calls"] >= state["max_calls"]:
            text = "not executed: the episode is over"
        else:
            text = world.call(name, args)
            state["calls"] += 1
            with log.open("a", encoding="utf-8") as f:
                f.write(json.dumps({"tool": name, "args": args}) + "\n")
        result = {"content": [{"type": "text", "text": text}], "isError": False}
    elif method == "ping":
        result = {}
    else:
        return {"jsonrpc": "2.0", "id": mid, "error": {"code": -32601, "message": f"unknown method {method}"}}
    return {"jsonrpc": "2.0", "id": mid, "result": result}


def main():
    sys.stdin.reconfigure(encoding="utf-8")
    sys.stdout.reconfigure(encoding="utf-8")
    ep = json.loads(Path(sys.argv[1]).read_text(encoding="utf-8"))
    log = Path(sys.argv[2])
    world, tools = make_world(ep["world"], ep["task"])
    state = {"calls": 0, "max_calls": ep.get("max_calls", 30)}
    for line in sys.stdin:
        if not line.strip():
            continue
        reply = handle(json.loads(line), world, tools, log, state)
        if reply is not None:
            sys.stdout.write(json.dumps(reply) + "\n")
            sys.stdout.flush()


if __name__ == "__main__":
    main()
```

- [ ] **Step 4: Implement `cli_backend.py`**

```python
"""Run one episode through the Claude Code CLI (subscription auth) instead of an API.

The CLI runs the tool loop itself; the world lives in an MCP server (mcp_world.py)
that logs every executed call. The caller replays those calls into a fresh world,
which is deterministic, so steps/final_ok/truth come out as with agent.run_episode.

Context differences against the API path, constant across Claude runs and reported
in the results: Claude Code adds a short account/workspace note to our system prompt,
and the model thinks by default. --setting-sources "" keeps settings hooks and
CLAUDE.md out; --tools "" removes the built-in tools. --bare would be cleaner but
disables subscription auth.
"""
from __future__ import annotations

import json
import shutil
import subprocess
import sys
import tempfile
from pathlib import Path

HERE = Path(__file__).parent
CLAUDE = shutil.which("claude") or "claude"


def build_cmd(model, system, user, mcp_cfg, max_turns):
    return [CLAUDE, "-p", user, "--system-prompt", system, "--model", model["model"],
            "--setting-sources", "", "--strict-mcp-config", "--mcp-config", str(mcp_cfg),
            "--tools", "", "--allowedTools", "mcp__world", "--no-session-persistence",
            "--output-format", "stream-json", "--verbose", "--max-turns", str(max_turns)]


def parse_stream(text):
    usage, result, model = {}, None, None
    for line in text.splitlines():
        try:
            m = json.loads(line)
        except json.JSONDecodeError:
            continue
        if m.get("type") == "system" and m.get("subtype") == "init":
            model = m.get("model")
        elif m.get("type") == "assistant":
            msg = m.get("message", {})
            usage[msg.get("id")] = msg.get("usage", {})  # one event per content block
        elif m.get("type") == "result":
            result = m
    return list(usage.values()), result, model


def run_cli_episode(model, world_name, task, system, user, max_calls=30, max_turns=40, timeout=900):
    """Returns (result dict, executed calls as [{"tool", "args"}])."""
    with tempfile.TemporaryDirectory() as tmp:
        d = Path(tmp)
        (d / "episode.json").write_text(json.dumps({"world": world_name, "task": task, "max_calls": max_calls}),
                                        encoding="utf-8")
        steps_log = d / "steps.jsonl"
        cfg = {"mcpServers": {"world": {"command": sys.executable,
                                        "args": [str(HERE / "mcp_world.py"), str(d / "episode.json"),
                                                 str(steps_log)]}}}
        (d / "mcp.json").write_text(json.dumps(cfg), encoding="utf-8")
        try:
            proc = subprocess.run(build_cmd(model, system, user, d / "mcp.json", max_turns), cwd=d,
                                  capture_output=True, text=True, encoding="utf-8", timeout=timeout)
        except subprocess.TimeoutExpired:
            return {"stop": "error", "error": f"timeout after {timeout}s", "calls": 0, "usage": []}, []
        calls = ([json.loads(l) for l in steps_log.read_text(encoding="utf-8").splitlines()]
                 if steps_log.exists() else [])
    usage, result, resolved = parse_stream(proc.stdout)
    if result is None:
        return {"stop": "error", "error": f"exit {proc.returncode}: {proc.stderr[-500:]}",
                "calls": len(calls), "usage": usage}, calls
    return {"stop": result.get("subtype"), "calls": len(calls), "usage": usage, "resolved_model": resolved,
            "num_turns": result.get("num_turns"), "final_text": result.get("result", "")}, calls
```

```python
"""Run agents over the tasks and append one JSON line per episode.

    uv run python run.py --models haiku,sonnet,qwen --world both --tasks 5 --reps 1 --out traces/pilot.jsonl

Claude models go through the Claude Code CLI (subscription); others through the API
loop in agent.py. Sequential on purpose (this machine saturates). Re-running with the
same --out skips episodes already recorded without error, so an interrupted run
resumes where it stopped.
"""
from __future__ import annotations

import argparse
import json
from pathlib import Path

import onboarding
import prompts
import rewrite
import transport
from agent import run_episode
from cli_backend import run_cli_episode

HERE = Path(__file__).parent
MAX_CALLS = 30


def load_models(names):
    configs = json.loads((HERE / "models.json").read_text())
    out = {}
    for n in names:
        m = dict(configs[n])
        if m.get("project") == "from-gcloud":
            m["project"] = transport.gcloud_project()
        out[n] = m
    return out


def make(world_name, task):
    if world_name == "rewrite":
        w = rewrite.RewriteWorld(task["start"], task["target"], task["assumptions"], kind=task["kind"])
        return w, prompts.REWRITE_SYSTEM, prompts.rewrite_user(task, w), rewrite.TOOLS
    w = onboarding.OnboardingWorld(task["customer"])
    user = prompts.ONBOARDING_USER.format(id=task["customer"]["id"])
    return w, prompts.ONBOARDING_SYSTEM, user, onboarding.TOOLS


def episode(model, wn, task):
    w, system, user, tools = make(wn, task)
    if model["transport"] != "claude_cli":
        return w, run_episode(model, w, system, user, tools, max_calls=MAX_CALLS)
    res, calls = run_cli_episode(model, wn, task, system, user, max_calls=MAX_CALLS)
    for c in calls:  # deterministic replay: same steps the MCP server produced
        w.call(c["tool"], c["args"])
    if res["stop"] != "error":
        res["stop"] = ("done" if w.done else "budget" if len(calls) >= MAX_CALLS
                       else "max_turns" if res["stop"] == "error_max_turns" else "no_tool_call")
    return w, res


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--models", required=True)
    ap.add_argument("--world", choices=["rewrite", "onboarding", "both"], default="both")
    ap.add_argument("--tasks", type=int, default=999, help="first N tasks of each world")
    ap.add_argument("--task-ids", default="", help="comma-separated ids; overrides --tasks")
    ap.add_argument("--reps", type=int, default=1)
    ap.add_argument("--out", required=True)
    args = ap.parse_args()

    out = Path(args.out)
    out.parent.mkdir(parents=True, exist_ok=True)
    done = set()
    if out.exists():
        for line in out.read_text(encoding="utf-8").splitlines():
            r = json.loads(line)
            if r["stop"] != "error":
                done.add((r["world"], r["task_id"], r["model"], r["rep"]))

    models = load_models(args.models.split(","))
    worlds = ["rewrite", "onboarding"] if args.world == "both" else [args.world]
    wanted = set(filter(None, args.task_ids.split(",")))
    for wn in worlds:
        tasks = json.loads((HERE / f"tasks/{wn}.json").read_text())
        tasks = [t for t in tasks if t["id"] in wanted] if wanted else tasks[:args.tasks]
        for task in tasks:
            for mname, model in models.items():
                for rep in range(args.reps):

                        continue
                    w, res = episode(model, wn, task)
                    rec = {"world": wn, "task_id": task["id"], "kind": task["kind"], "model": mname,
                           "rep": rep, **res, "steps": w.steps,
                           "final_ok": w.final_ok(), "truth": w.truth()}
                    with out.open("a", encoding="utf-8") as f:
                        f.write(json.dumps(rec) + "\n")
                    print(f"{wn} {task['id']} {mname} r{rep}: stop={res['stop']} calls={res['calls']} "
                          f"final_ok={rec['final_ok']} truth={rec['truth']}"
                          + (f" error={res['error'][:200]}" if res.get("error") else ""), flush=True)


if __name__ == "__main__":
    main()

"""Score every trace with E0-E3 against the semantic truth.

    uv run python analyze.py traces/pilot.jsonl [--pilot]

Groups: rewrite (solvable), rewrite-trap, onboarding. Counts are absolute: n is small.
"""
from __future__ import annotations

import argparse
import json
from pathlib import Path

from evaluators import condensed, e0, e1, e2, e3

HERE = Path(__file__).parent
TAUS = [0.5, 0.6, 0.7, 0.8, 0.9, 1.0]


def load(path):
    last = {}
    for line in Path(path).read_text(encoding="utf-8").splitlines():
        r = json.loads(line)
        last[(r["world"], r["task_id"], r["model"], r["rep"])] = r
    return [r for r in last.values() if r["stop"] != "error"]


def load_tasks():
    out = {}
    for wn in ("rewrite", "onboarding"):
        for t in json.loads((HERE / f"tasks/{wn}.json").read_text()):
            out[(wn, t["id"])] = t
    return out


def group(r):
    return "rewrite-trap" if r["kind"] == "trap" else r["world"]


def score(r, task, tau):
    return {"E0": e0(r), "E1": e1(r, task["rules"]), "E2": e2(r, task["rules"]),
            "E3": e3(r, task["reference_tokens"], tau)}


def table(traces, tasks, tau=0.8):
    rows = {}
    for r in traces:
        for ev, v in score(r, tasks[(r["world"], r["task_id"])], tau).items():
            if v is None:
                continue
            c = rows.setdefault((group(r), ev), {"n": 0, "FP": 0, "FN": 0})
            c["n"] += 1
            c["FP"] += int(v and not r["truth"])
            c["FN"] += int(not v and r["truth"])
    return rows


def fn_kind(tokens, refs):
    for ref in refs:
        if sorted(tokens) == sorted(ref):
            return "order"
        it = iter(tokens)
        if all(t in it for t in ref):
            return "extra_steps"
    return "other"


def tokens_used(r):
    total = 0
    for u in r.get("usage", []):
        total += u.get("input_tokens", 0) + u.get("output_tokens", 0)
        total += u.get("prompt_tokens", 0) + u.get("completion_tokens", 0)
    return total


def print_table(rows):
    print(f"{'group':<14}{'eval':<6}{'n':>5}{'FP':>5}{'FN':>5}")
    for (g, ev), c in sorted(rows.items()):
        print(f"{g:<14}{ev:<6}{c['n']:>5}{c['FP']:>5}{c['FN']:>5}")


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("traces")
    ap.add_argument("--pilot", action="store_true")
    args = ap.parse_args()
    traces, tasks = load(args.traces), load_tasks()

    print(f"{len(traces)} traces\n")
    print_table(table(traces, tasks))

    print("\nE3 by tau (FP/FN):")
    for g in ("rewrite", "onboarding"):
        row = []
        for tau in TAUS:
            c = table([r for r in traces if group(r) == g], tasks, tau).get((g, "E3"), {"FP": 0, "FN": 0})
            row.append(f"{tau}:{c['FP']}/{c['FN']}")
        print(f"  {g:<12}", "  ".join(row))

    print("\nE3 false negatives by kind (tau=0.8):")
    for r in traces:
        task = tasks[(r["world"], r["task_id"])]
        if r["truth"] and e3(r, task["reference_tokens"], 0.8) is False:
            print(f"  {r['world']} {r['task_id']} {r['model']} r{r['rep']}: "
                  f"{fn_kind(condensed(r['steps']), task['reference_tokens'])}")

    print("\nE2 disagreements with the truth (declared rules vs semantics):")
    for r in traces:
        v = e2(r, tasks[(r["world"], r["task_id"])]["rules"])
        if v != r["truth"]:
            print(f"  {'FP' if v else 'FN'} {r['world']} {r['task_id']} {r['model']} r{r['rep']}")

    print("\nExamples: final state right, derivation false (E0 FP):")
    for r in traces:
        if r["final_ok"] and not r["truth"]:
            print(f"  {r['world']} {r['task_id']} {r['model']} r{r['rep']}")

    if args.pilot:
        rw = [r for r in traces if r["world"] == "rewrite"]
        invalid = sum(any(s.get("valid") is False for s in r["steps"]) for r in rw)
        detours = 0
        for r in rw:
            refs = tasks[("rewrite", r["task_id"])]["reference_tokens"]
            if r["truth"] and refs and len(condensed(r["steps"])) > min(map(len, refs)):
                detours += 1
        per_model = {}
        for r in traces:
            per_model.setdefault(r["model"], []).append(tokens_used(r))
        print(f"\nPILOT: rewrite traces with an invalid step = {invalid} (need >= 3)")
        print(f"PILOT: valid rewrite traces longer than the shortest reference = {detours} (need >= 3)")
        for m, xs in per_model.items():
            print(f"PILOT: {m} mean tokens per trace = {sum(xs) // max(1, len(xs))}")


if __name__ == "__main__":
    main()
```
